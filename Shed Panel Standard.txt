# -------------------------------------------------------------
# HLK-LD2410C mmWave presence sensor
#
#   LD2410C GND -> GND
#   LD2410C TX  -> GPIO26  (ESP32 RX)
#   LD2410C RX  -> GPIO27  (ESP32 TX)
#
# Avoid GPIO16/17 - the WROVER-B uses those for PSRAM.
# GPIO34/35/36/39 are input-only: usable as RX, never as TX.
#
# The module's default baud is 256000. If you get no data at all,
# it has probably been reconfigured - try 115200 or 9600.
# -------------------------------------------------------------

binary_sensor:
  - platform: homeassistant
    id: s_doorbell
    entity_id: ${doorbell}
    on_press:
      - light.turn_on:
          id: backlight
          brightness: 100%
      - lvgl.page.show: page_cam
      - lvgl.widget.update: {id: nav_home, state: {checked: false}}
      - lvgl.widget.update: {id: nav_cam, state: {checked: true}}
      - lambda: 'id(img_door)->update();'
      - script.execute: restore_panel
        
  - platform: ld2410
    has_target:
      name: Presence
    has_moving_target:
      name: Moving Target
    has_still_target:
      name: Still Target
      
script:
  # mode: restart means repeated doorbell presses keep pushing the
  # timeout out rather than stacking up
  - id: restore_panel
    mode: restart
    then:
      - delay: 90s
      - lvgl.page.show: page_home
      - lvgl.widget.update: {id: nav_cam, state: {checked: false}}
      - lvgl.widget.update: {id: nav_home, state: {checked: true}}
      - if:
          condition:
            lambda: 'return id(is_night);'
          then:
            - light.turn_on:
                id: backlight
                brightness: 12%
