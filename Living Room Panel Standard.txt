# Front door camera
#
# Point this at a PRE-RESIZED image ~464px wide. go2rtc:
#   http://<ha-ip>:1984/api/frame.jpeg?src=front_door&w=464
# -------------------------------------------------------------
image:
  - platform: online_image
    id: img_door
    url: http://IP:1984/api/frame.jpeg?src=front_door&w=464
    format: JPEG
    type: RGB565
    update_interval: never
    # Without this the widget keeps drawing the empty buffer it
    # cached at boot, even though the download succeeds.
    on_download_finished:
      - lvgl.image.update:
          id: cam_view
          src: img_door
    on_error:
      - logger.log:
          level: WARN
          format: "Door snapshot failed"
# Refresh the door snapshot only while the camera page is showing.
# Fetch + decode blocks the main loop, so polling this on other
# pages would make the whole UI feel sluggish for no benefit.
interval:
  - interval: 2s
    then:
      - if:
          condition:
            lvgl.page.is_showing: page_cam
          then:
            - lambda: 'id(img_door)->update();'
